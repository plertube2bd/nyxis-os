; =========================================================
; Nyxis OS Syscall ABI v2 (NX_ABI_VERSION 2.4)
; x86_64
; =========================================================

; rax = syscall number
; rdi, rsi, rdx, r10, r8, r9 = args
; return:
;   rax >= 0  : success
;   rax < 0   : error (Nstatus, i.e. include/types.h Nerror, sign-extended to 64 bits;
;               NOT a POSIX -errno value)
;
; implemented so far (include/nyx_abi.h 에 번호/구조체/상수가 정의되어 있다. include/nyx_abi.h 는
; 커널과 유저랜드가 함께 포함하는 공개 ABI 헤더이며, 이 파일의 번호가 바뀌면 항상 함께 갱신한다):
;
;   2  NxGetVersion            7  NxSysInfo(struct nx_sysinfo *out)
;   3  NxGetTime(clock_id)                clock_id: NX_CLOCK_MONOTONIC(0), NX_CLOCK_REALTIME(1, CMOS RTC)
;   4  NxSleep(ms)                        ms <= NX_SLEEP_MAX_MS(1h)
;   5  NxYield
;   8  NxGetIdentity(struct nx_identity *out, token)   uid/gid/사용자 이름. token=NX_NO_TOKEN(0) 이면 자신,
;                                      토큰 핸들(NX_RIGHT_STAT)이면 그 토큰의 신원 (소모하지 않음)
;   9  NxPowerOff()                       uid 0 만 (보통 init 이 svc:/sys/init 요청을 받아 호출)
;   10 NxAuthenticate(name, password)     -> 일회용 토큰 핸들. 실패 시 1초 지연 후 Nauthentication
;
;   32 NxOpen(const char *path, flags)    flags: NX_O_READ|NX_O_WRITE|NX_O_APPEND -> 핸들
;                                      NX_O_APPEND: 파일은 쓰기마다 끝으로 (열린 파일의 속성: dup/fork/exec 에 이어짐)
;                                      info:/ 는 읽기만 (쓰기 열기는 Npermission), 열 때의 스냅샷
;   33 NxClose(handle)
;   34 NxRead(handle, void *buf, len)     -> 읽은 바이트 수 (0 = EOF). stdin(콘솔)은 줄 단위:
;                                          한 줄이 완성될 때까지 잠들고 최대 한 줄을 돌려준다.
;                                          Ctrl-D = EOF (console/inputs/tty.h)
;   35 NxWrite(handle, const void *buf, len) -> 쓴 바이트 수. 콘솔(stdout/stderr, dev:/tty)은 터미널 장치
;                                          (console/inputs/tty.c 의 tty_write)로 바로 간다 - printk 를 거치지 않는다
;   36 NxSeek(handle, offset, whence)     whence: NX_SEEK_SET/CUR/END -> 새 오프셋
;   37 NxStat(handle, struct nx_stat *out)
;   41 NxDuplicateHandle(handle, rights_mask) -> 새 핸들 (권한은 줄이기만 가능)
;   42 NxPoll(struct nx_pollfd *fds, n, timeout_ms) -> 준비된 수 (채널/파이프/프로세스/stdin, 레벨 트리거)
;   43 NxDuplicateHandleEx(handle, rights, flags, target) NX_DUP_SHARE(열린 파일 공유) / TO_SLOT(dup2) /
;                                          MIN_SLOT(F_DUPFD) / CLOEXEC
;   44 NxHandleFlags(handle, set, clear) -> 플래그 (NX_HFLAG_CLOEXEC)
;
;   경로는 이름공간 규칙을 따른다: fs:/ (root 만), app:/=fs:/usr/, face:/=fs:/home/<사용자>/,
;   roi:/=fs:/roi/ (root 만, roi:/users 는 커널 전용), tmp:/, dev:/ (root 만, 단 표준 스트림 장치
;   dev:/stdin stdout stderr tty 는 누구나), svc:/ (채널)
;
;   64 NxProcessCreate(const char *path, argv, envp, token) -> 프로세스 핸들 (WAIT|KILL|STAT|DUP).
;                                          token: NX_NO_TOKEN(0) 이면 부모 신원 상속, 아니면 그 토큰의
;                                          신원으로 시작하고 토큰은 소모된다.
;                                          ELF64 ET_EXEC 를 새 주소 공간에 적재하고 System V
;                                          초기 스택(argc/argv/envp/auxv)으로 ring3 실행.
;                                          자세한 규약은 include/nyx_abi.h 참고.
;   65 NxProcessExit(exit_code)           반환하지 않음
;   66 NxProcessKill(handle)              NX_RIGHT_KILL. 종료 코드 NX_EXIT_KILLED(-2)
;   67 NxProcessWait(handle, timeout_ms, int32_t *out_code)  NX_RIGHT_WAIT. 잠들어 기다림.
;                                          NX_WAIT_INFINITE / 0 = 확인만 / 그 외 ms (Ntimeout)
;   68 NxProcessInfo(NX_PID_SELF 또는 프로세스 핸들, struct nx_procinfo *out)   parent_pid 포함
;   64 의 다섯 번째 인자 flags: NX_SPAWN_INHERIT = CLOEXEC 가 아닌 핸들을 같은 슬롯으로 물려준다
;   73 NxProcessFork() -> 부모: 자식 핸들, 자식: 0   (실행 계층 kernel/exec: 주소 공간 즉시 복사,
;                                          핸들은 슬롯/세대 그대로 상속, 열린 파일 공유)
;   74 NxProcessExec(path, argv, envp)    성공하면 돌아오지 않는다. 새 이미지를 다 만든 뒤 교체.
;                                          CLOEXEC 핸들 닫음, 남은 핸들 세대 0. 실행 권한(x) 필요
;   75 NxProcessGroup(NX_PID_SELF 또는 프로세스 핸들, pgid) -> 그룹 번호. pgid 0 = 새 그룹(대상 pid),
;                                          NX_PGID_QUERY = 조회. Ctrl-C 는 콘솔의 포그라운드 그룹
;                                          (NX_DEVCMD_TTY_FOREGROUND)에 SIGINT 를 보낸다
;   76 NxSignalAction(sig, act, old)      처리기 등록 (struct nx_sigaction: handler, restorer, mask, flags)
;   77 NxSignalMask(how, set) -> 이전 집합  NX_SIG_BLOCK / UNBLOCK / SETMASK
;   78 NxSignalSend(NX_PID_SELF 또는 프로세스 핸들, sig)   핸들은 NX_RIGHT_KILL. SIGKILL = NxProcessKill
;   79 NxSignalReturn()                   처리기에서 복귀 (사용자 스택의 nx_sigcontext 로, iretq)
;   80 NxSignalSuspend(set)               막은 집합을 잠시 바꾸고 신호를 기다린다. 늘 Ninterrupted
;   81 NxSignalPending() -> 막혀서 대기 중인 집합
;                                          전달 규칙/스택 모양: include/nyx_abi.h 의 "신호", kernel/process/signal.h
;
;   96 NxVirtualAlloc(increment)          -> 새 brk. 부호 있는 증가분(sbrk 방식): 양수=늘리기,
;                                          음수=줄이기, 0=조회. 97(NxVirtualFree)은 미래의
;                                          mmap 스타일 API 를 위해 번호만 예약(v1 미구현).
;
;   128 NxPipeCreate(u64 out[2], flags)   IPC 계층 kernel/ipc/pipe: 읽기/쓰기 끝, 블로킹, EOF/Ndisconnected
;   129 NxChannelCreate(path, flags)      svc:/sys/<이름> (root) 또는 svc:/<사용자>/<이름> -> 수신 핸들
;   130 NxChannelSend(h, buf, len)        NxOpen("svc:/...", NX_O_WRITE) 로 얻은 송신 핸들. 가득 차면 Nbusy
;   131 NxChannelReceive(h, buf, len, struct nx_msginfo *info, timeout_ms) 보낸 쪽 pid/uid/gid 가 붙는다
;
;   162 NxCreateNode(path, type, mode)    163 NxRemoveNode(path)
;   164 NxRenameNode(old, new)            165 NxReadDir(path, index, struct nx_dirent *out)
;       NxReadDir("dev:/") 는 누구나: stdin stdout stderr tty, 그다음 (root 에게만) 등록된 장치 이름
;   162 의 mode 에는 호출자의 umask 가 적용된다
;   166 NxStatPath(path, struct nx_stat *out)   보안 계층 kernel/security/perm: 열지 않고 (mode/uid/gid 포함)
;   167 NxSetMode(path, mode)             chmod. 소유자 또는 uid 0, mode <= 0777
;   168 NxSetUmask(mask) -> 이전 값
;
;   193 NxDeviceCommand(h, cmd, arg)      NX_DEVCMD_INFO / IOPERM / TTY_ECHO(stdin)
;   194 NxDeviceMapMMIO(h, u64 *out_addr) dev:/fb0 를 자기 주소 공간에 매핑
;       장치는 NxOpen("dev:/fb0|com1|rd0|rootdisk") 로 연다 (root 만, 배타적 소유)
;       표준 스트림 장치 (누구나, 공유): dev:/tty = 콘솔 터미널, dev:/stdin|stdout|stderr = 호출자의 0/1/2 번이
;       가리키는 객체를 공유하는 새 핸들 (리디렉션을 따른다. 닫혀 있으면 NnotFound, 없는 권한은 Npermission)
;
;   771 NxKernelPrint(const char *str)    디버그 로그. 서식 문자열 아님, 제어 문자는 '?' 로 치환
;   777 NxDebugNop
;
; 그 외 번호는 NsyscallFailed. 사용자 포인터는 모두 검증된다 (uaccess.h: 페이지 테이블 USER/RW 비트 확인 +
; 접근 도중 매핑이 바뀌는 경쟁에도 안전한 폴트 복구 복사). 핸들은 프로세스별 테이블에서 권한(NX_RIGHT_*)까지
; 함께 검사한다(handles.h). 구현은 nxkernel/kernel/syscall/syscall.c 의 g_syscalls 표를 참고.
;
; 프로세스별 독립 주소 공간(kernel/paging/paging.h 의 addr_space_t), 물리 프레임 할당자
; (kernel/mm/pfa.h), ELF64 로더(kernel/process/elf.h), 힙/brk(kernel/process/umem.h) 는
; 모두 NxProcessCreate/NxVirtualAlloc 을 뒷받침하는 내부 구현이며 그 자체로 ABI 는 아니다.
;
; errno: 커널은 POSIX errno 를 모른다. Nstatus -> errno 변환은 nlibc 가 소유한다
; (userland/nlibc/library/errno.h 의 결정 사항 주석 참고).
;
; object handle type:
;   u64 handle (상위 32비트 = 세대, 하위 32비트 = 슬롯. 닫힌 핸들 값의 재사용을 막는다)
;
; namespace style:
;   Nx*
